#### 1. `/src/api/search.ts`

```typescript
import request from '@/utils/request'

// 搜索结果接口
export interface SearchResult {
  resourceType: string
  resourceId: string
  title: string
  subtitle: string
  description: string
  highlightSnippet: string
  score: number
  status: number
  createdAt: string
  updatedAt: string
}

// 搜索参数
export interface SearchParams {
  keyword: string
  types?: string[]
  page?: number
  size?: number
}

// 搜索API
export function search(params: SearchParams) {
  return request({
    url: '/admin-api/v1/search',
    method: 'get',
    params: {
      keyword: params.keyword,
      types: params.types?.join(','),
      page: params.page || 1,
      size: params.size || 20
    }
  })
}

// 重建索引
export function rebuildIndex(type?: string) {
  return request({
    url: '/admin-api/v1/search/rebuild',
    method: 'post',
    params: { type }
  })
}
```

### 二、全局搜索组件（1个文件）

#### `/src/components/GlobalSearch/index.vue`

```vue
<template>
  <div class="global-search">
    <el-popover
      v-model:visible="visible"
      :width="600"
      trigger="click"
      placement="bottom-start"
    >
      <template #reference>
        <el-input
          v-model="keyword"
          placeholder="搜索..."
          :prefix-icon="Search"
          @keyup.enter="handleSearch"
          @focus="visible = true"
          style="width: 300px"
        />
      </template>
      
      <div v-loading="loading" class="search-results">
        <div v-if="results.length === 0 && !loading" class="no-data">
          {{ keyword ? '暂无搜索结果' : '请输入关键词搜索' }}
        </div>
        
        <div v-else class="result-list">
          <div
            v-for="item in results"
            :key="item.resourceId"
            class="result-item"
            @click="handleClick(item)"
          >
            <div class="result-header">
              <el-tag :type="getTagType(item.resourceType)" size="small">
                {{ getTypeName(item.resourceType) }}
              </el-tag>
              <span class="result-title">{{ item.title }}</span>
            </div>
            <div v-if="item.subtitle" class="result-subtitle">
              {{ item.subtitle }}
            </div>
            <div v-if="item.description" class="result-description">
              {{ item.description }}
            </div>
          </div>
        </div>
        
        <div v-if="total > pageSize" class="result-footer">
          <el-pagination
            v-model:current-page="page"
            :page-size="pageSize"
            :total="total"
            layout="prev, pager, next"
            small
            @current-change="handleSearch"
          />
        </div>
      </div>
    </el-popover>
  </div>
</template>

<script setup lang="ts">
import { ref, watch } from 'vue'
import { Search } from '@element-plus/icons-vue'
import { search, SearchResult } from '@/api/search'
import { ElMessage } from 'element-plus'

const visible = ref(false)
const keyword = ref('')
const loading = ref(false)
const results = ref<SearchResult[]>([])
const page = ref(1)
const pageSize = ref(10)
const total = ref(0)

// 防抖搜索
let searchTimer: any = null
watch(keyword, (val) => {
  if (!val) {
    results.value = []
    total.value = 0
    return
  }
  
  clearTimeout(searchTimer)
  searchTimer = setTimeout(() => {
    page.value = 1
    handleSearch()
  }, 300)
})

const handleSearch = async () => {
  if (!keyword.value) return
  
  loading.value = true
  try {
    const res = await search({
      keyword: keyword.value,
      page: page.value,
      size: pageSize.value
    })
    results.value = res.data.list
    total.value = res.data.total
  } catch (error) {
    ElMessage.error('搜索失败')
  } finally {
    loading.value = false
  }
}

const handleClick = (item: SearchResult) => {
  // 根据资源类型跳转到对应页面
  const routes: Record<string, string> = {
    user: '/system/user',
    role: '/system/role',
    menu: '/system/menu',
    dept: '/system/dept',
    dict: '/system/dict',
    config: '/system/config'
  }
  
  const route = routes[item.resourceType]
  if (route) {
    window.location.href = `${route}?id=${item.resourceId}`
  }
  visible.value = false
}

const getTypeName = (type: string) => {
  const names: Record<string, string> = {
    user: '用户',
    role: '角色',
    menu: '菜单',
    dept: '部门',
    dict: '字典',
    config: '配置'
  }
  return names[type] || type
}

const getTagType = (type: string) => {
  const types: Record<string, any> = {
    user: 'success',
    role: 'warning',
    menu: 'info',
    dept: 'primary',
    dict: 'danger',
    config: ''
  }
  return types[type] || ''
}
</script>
```

```vue
        <el-descriptions-item label="级别">{{ currentLog.level }}</el-descriptions-item>
        <el-descriptions-item label="Logger">{{ currentLog.logger }}</el-descriptions-item>
        <el-descriptions-item label="线程">{{ currentLog.thread }}</el-descriptions-item>
        <el-descriptions-item label="消息">{{ currentLog.message }}</el-descriptions-item>
        <el-descriptions-item v-if="currentLog.exception" label="异常堆栈">
          <pre style="max-height: 300px; overflow-y: auto">{{ currentLog.exception }}</pre>
        </el-descriptions-item>
        <el-descriptions-item v-if="currentLog.traceId" label="TraceID">

        </el-descriptions-item>
      </el-descriptions>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { queryRuntimeLogs, RuntimeLog } from '@/api/log'
import { ElMessage } from 'element-plus'

const loading = ref(false)
const logList = ref<RuntimeLog[]>([])
const timeRange = ref<string[]>([])
const queryParams = ref({
  level: '',
  logger: '',
  message: '',
  limit: 100
})

const dialogVisible = ref(false)
const currentLog = ref<RuntimeLog>({} as RuntimeLog)

onMounted(() => {
  // 默认查询最近1小时
  const now = new Date()
  const oneHourAgo = new Date(now.getTime() - 60 * 60 * 1000)
  timeRange.value = [
    formatDateTime(oneHourAgo),
    formatDateTime(now)
  ]
  handleQuery()
})

const handleQuery = async () => {
  loading.value = true
  try {
    const res = await queryRuntimeLogs({
      ...queryParams.value,
      startTime: timeRange.value?.[0],
      endTime: timeRange.value?.[1]
    })
    logList.value = res.data
  } catch (error) {
    ElMessage.error('查询失败')
  } finally {
    loading.value = false
  }
}

const handleReset = () => {
  queryParams.value = {
    level: '',
    logger: '',
    message: '',
    limit: 100
  }
  timeRange.value = []
  handleQuery()
}

const handleDetail = (row: RuntimeLog) => {
  currentLog.value = row
  dialogVisible.value = true
}

const getLevelType = (level: string) => {
  const types: Record<string, any> = {
    ERROR: 'danger',
    WARN: 'warning',
    INFO: 'info',
    DEBUG: ''
  }
  return types[level] || ''
}

const formatDateTime = (date: Date) => {
  return date.toISOString().slice(0, 19).replace('T', ' ')
}
</script>
```
