```java
/**
 * 测试 JWT 配置。
 *
 * <p>使用当前配置生成测试 Token 并验证。
 *
 * @param request 测试请求
 * @return 测试响应
 */
public Mono<JwtTestResponse> testJwt(JwtTestRequest request) {
    return getByTenantId()
            .flatMap(config -> Mono.fromCallable(() -> {
                        try {
                            // 1. 生成 Token
                            Algorithm algorithm = getAlgorithm(config);
                            JWTCreator.Builder builder = JWT.create()
                                    .withIssuer(config.getIssuer());

                            // 添加 Payload
                            request.getPayload().forEach((key, value) -> {
                                if (value instanceof String) {
                                    builder.withClaim(key, (String) value);
                                } else if (value instanceof Integer) {
                                    builder.withClaim(key, (Integer) value);
                                } else if (value instanceof Long) {
                                    builder.withClaim(key, (Long) value);
                                } else if (value instanceof Boolean) {
                                    builder.withClaim(key, (Boolean) value);
                                }
                            });

                            // 设置过期时间
                            long expiresIn = request.getExpiresIn() != null 
                                    ? request.getExpiresIn() 
                                    : config.getAccessTokenTtl();
                            LocalDateTime expiresAt = LocalDateTime.now().plusSeconds(expiresIn);
                            builder.withExpiresAt(Date.from(expiresAt.atZone(ZoneId.systemDefault()).toInstant()));

                            LocalDateTime issuedAt = LocalDateTime.now();
                            builder.withIssuedAt(Date.from(issuedAt.atZone(ZoneId.systemDefault()).toInstant()));

                            String token = builder.sign(algorithm);

                            // 2. 解码验证
                            JWTVerifier verifier = JWT.require(algorithm)
                                    .withIssuer(config.getIssuer())
                                    .build();
                            DecodedJWT decoded = verifier.verify(token);

                            Map<String, Object> decodedPayload = new HashMap<>();
                            decoded.getClaims().forEach((key, claim) -> {
                                if (!claim.isNull()) {
                                    decodedPayload.put(key, claim.asString());
                                }
                            });

                            return JwtTestResponse.builder()
                                    .token(token)
                                    .decoded(decodedPayload)
                                    .isValid(true)
                                    .expiresAt(expiresAt)
                                    .issuedAt(issuedAt)
                                    .issuer(config.getIssuer())
                                    .build();

                        } catch (Exception e) {
                            log.error("JWT 测试失败", e);
                            return JwtTestResponse.builder()
                                    .isValid(false)
                                    .build();
                        }
                    })
                    .subscribeOn(Schedulers.boundedElastic()));
}

/**
 * 获取签名算法。
 *
 * @param config JWT 配置
 * @return 签名算法
 */
private Algorithm getAlgorithm(JwtConfigDTO config) {
    return switch (config.getAlgorithm().toUpperCase()) {
        case "HS256" -> Algorithm.HMAC256(config.getSecret());
        case "HS512" -> Algorithm.HMAC512(config.getSecret());
        case "RS256" -> {
            // RSA 算法需要公钥私钥
            // TODO: 实现 RSA 算法支持
            throw new BusinessException(ResultCode.ERROR, "暂不支持 RSA 算法");
        }
        default -> throw new BusinessException(ResultCode.ERROR, "不支持的签名算法: " + config.getAlgorithm());
    };
}
```
