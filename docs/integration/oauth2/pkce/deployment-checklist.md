## 附录

### A. 快速检查命令

```bash
#!/bin/bash
# pkce-deployment-check.sh

echo "=========================================="
echo "PKCE 功能部署快速检查脚本"
echo "=========================================="

# 1. 单元测试
echo "[1/7] 运行单元测试..."
mvnd test -Dtest="OAuth2AuthorizationServiceImplTest" -pl apex-application
if [ $? -ne 0 ]; then
    echo "❌ 单元测试失败"
    exit 1
fi
echo "✅ 单元测试通过"

# 2. 检查环境变量
echo "[2/7] 检查环境变量..."
if [ -z "$JWT_SECRET" ]; then
    echo "❌ JWT_SECRET 未设置"
    exit 1
fi
echo "✅ 环境变量检查通过"

# 3. 检查 Redis 连接
echo "[3/7] 检查 Redis 连接..."
redis-cli -h ${REDIS_HOST:-localhost} -p ${REDIS_PORT:-6379} ping > /dev/null
if [ $? -ne 0 ]; then
    echo "❌ Redis 连接失败"
    exit 1
fi
echo "✅ Redis 连接正常"

# 4. 检查硬编码密钥
echo "[4/7] 检查硬编码密钥..."
HARDCODED=$(grep -r "secret.*=.*\"" --include="*.java" | wc -l)
if [ $HARDCODED -gt 0 ]; then
    echo "⚠️  发现疑似硬编码密钥，请人工检查"
fi
echo "✅ 硬编码检查完成"

# 5. 构建检查
echo "[5/7] 执行构建..."
mvnd clean package -DskipTests -pl apex-application
if [ $? -ne 0 ]; then
    echo "❌ 构建失败"
    exit 1
fi
echo "✅ 构建成功"

# 6. 检查文档
echo "[6/7] 检查文档完整性..."
DOCS=(
    ".trae/documents/pkce_verification_plan.md"
    ".trae/documents/pkce_unit_test_report.md"
    ".trae/documents/pkce_integration_test_report_v2.md"
    ".trae/documents/pkce_performance_test_report.md"
    ".trae/documents/pkce_e2e_test_script.md"
)
for doc in "${DOCS[@]}"; do
    if [ ! -f "$doc" ]; then
        echo "⚠️  文档缺失: $doc"
    fi
done
echo "✅ 文档检查完成"

# 7. 总结
echo "[7/7] 检查完成"
echo "=========================================="
echo "✅ 快速检查通过，可进行详细检查"
echo "=========================================="
```
